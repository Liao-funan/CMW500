Specifying Measurement-Specific Settings
// *****************************************************************************
// Define 2 segments with a length of 20 bursts each
// and different modulation type.
// *****************************************************************************
CONFigure:WLAN:MEAS:MEValuation:LIST:COUNt 2
CONFigure:WLAN:MEAS:MEValuation:LIST:CAPTime 0.002
CONFigure:WLAN:MEAS:MEValuation:LIST:SEGMent1:SETup 20, LOFDm, BPM6, LONG
CONFigure:WLAN:MEAS:MEValuation:LIST:SEGMent2:SETup 20, LOFDm, QM12, LONG
 
// *****************************************************************************
// Enable modulation and transmit spectrum mask results for both segments.
// Select an averaging length of 20 (all measured bursts in the segment).
// Enable the list mode.
// *****************************************************************************
CONFigure:WLAN:MEAS:MEValuation:LIST:SEGMent1:MODulation 20,ON
CONFigure:WLAN:MEAS:MEValuation:LIST:SEGMent2:MODulation 20,ON
CONFigure:WLAN:MEAS:MEValuation:LIST:SEGMent1:TSMask 20,ON 
CONFigure:WLAN:MEAS:MEValuation:LIST:SEGMent2:TSMask 20,ON
CONFigure:WLAN:MEAS:MEValuation:LIST ON
 
// *****************************************************************************
// Use a power trigger to start the measurement.
// *****************************************************************************
